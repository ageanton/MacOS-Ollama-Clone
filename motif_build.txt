#!/usr/bin/env bash
# Small launcher. Never needs editing.
# Put the FULL workflow text (name: Build Motif ... steps: ...) in motif_full.txt.
# This converts it into a bash script and runs it.
set -e
python3 - <<'PY'
import os, re, shlex, sys
src = "motif_full.txt"
if not os.path.exists(src):
    sys.exit("motif_full.txt not found in the repo root")
text = open(src, encoding="utf-8").read()
lines = text.split("\n")
marker = " " * 4 + "steps:"
if not any(l.rstrip() == marker for l in lines):
    script = text
else:
    out = ["#!/usr/bin/env bash", "set -e"]
    in_env = False
    for l in lines:
        if re.match(r"^env:\s*$", l):
            in_env = True
            continue
        if in_env:
            m = re.match(r"^  ([A-Za-z_][A-Za-z0-9_]*):\s*(.*?)\s*$", l)
            if m:
                v = m.group(2)
                if len(v) >= 2 and v[0] == v[-1] and v[0] in "\"'":
                    v = v[1:-1]
                out.append("export %s=%s" % (m.group(1), shlex.quote(v)))
            elif l.strip() and not l.startswith(" "):
                in_env = False
    i = next(k for k, l in enumerate(lines) if l.rstrip() == marker) + 1
    n = len(lines)
    count = 0
    while i < n:
        m = re.match(r"^      - name:\s*(.*)$", lines[i])
        if not m:
            i += 1
            continue
        name = m.group(1).strip()
        i += 1
        wd = None
        body = None
        while i < n and not re.match(r"^      - ", lines[i]):
            l = lines[i]
            w = re.match(r"^        working-directory:\s*(\S+)", l)
            if w:
                wd = w.group(1)
            if re.match(r"^        run:\s*\|[-+]?\s*$", l):
                i += 1
                body = []
                while i < n and (not lines[i].strip() or lines[i].startswith(" " * 10)):
                    body.append(lines[i][10:] if lines[i].strip() else "")
                    i += 1
                continue
            i += 1
        if body is not None:
            count += 1
            out.append("printf '::group::%s\\n' " + shlex.quote(name))
            out.append("(")
            out.append("set -e")
            if wd:
                out.append("cd " + shlex.quote(wd))
            out.extend(body)
            out.append(")")
            out.append("echo '::endgroup::'")
    if count == 0:
        sys.exit("No run steps found in motif_full.txt")
    print("Extracted %d build steps" % count)
    script = "\n".join(out) + "\n"
target = os.path.join(os.environ.get("RUNNER_TEMP", "/tmp"), "motif_run.sh")
open(target, "w", encoding="utf-8").write(script)
PY
bash "${RUNNER_TEMP:-/tmp}/motif_run.sh"
